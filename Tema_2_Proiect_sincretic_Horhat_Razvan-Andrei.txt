#include <LiquidCrystal.h>

// ---------- LCD ----------
const uint8_t PIN_RS = 12;
const uint8_t PIN_E  = 11;
const uint8_t PIN_D4 = 5;
const uint8_t PIN_D5 = 4;
const uint8_t PIN_D6 = 3;
const uint8_t PIN_D7 = 2;
LiquidCrystal lcd(PIN_RS, PIN_E, PIN_D4, PIN_D5, PIN_D6, PIN_D7);

// ---------- Ora initiala (setata de noi) ----------
volatile uint8_t ore  = 10;
volatile uint8_t min  = 10;
volatile uint8_t sec  = 10;

// Flag setat in ISR, folosit in loop()
volatile bool tickSecunda = false;

// ---------- ADC ----------
// LM35: 10 mV / grad Celsius
// Referinta interna 1.1V -> 1 LSB = 1100mV / 1024 ~ 1.074 mV -> 0.1074 grade
void adcInit() {
  // REFS1:REFS0 = 11 -> referinta interna 1.1V, canal ADC0 (A0)
  ADMUX = (1 << REFS1) | (1 << REFS0);
  // ADEN: activare ADC, prescaler 128 (16MHz/128 = 125kHz)
  ADCSRA = (1 << ADEN) | (1 << ADPS2) | (1 << ADPS1) | (1 << ADPS0);
}

uint16_t adcCitire() {
  ADCSRA |= (1 << ADSC);              // porneste conversia
  while (ADCSRA & (1 << ADSC)) {}     // asteapta finalul conversiei
  return ADC;
}

float citesteTemperatura() {
  // prima citire dupa schimbarea referintei se ignora
  adcCitire();
  uint32_t suma = 0;
  for (uint8_t i = 0; i < 16; i++) {  // media a 16 esantioane
    suma += adcCitire();
  }
  float adc = suma / 16.0;
  float mV = adc * 1100.0 / 1024.0;
  // LM35  : T = mV / 10
  // TMP36 : T = (mV - 500) / 10   (varianta din Tinkercad, offset 500 mV)
  return (mV - 500.0) / 10.0;
}

// ---------- Timer1: intrerupere la 1 secunda ----------
// 16 MHz / 1024 = 15625 Hz  ->  OCR1A = 15625 - 1 = 15624  -> 1 Hz
void timer1Init() {
  noInterrupts();
  TCCR1A = 0;
  TCCR1B = 0;
  TCNT1  = 0;
  OCR1A  = 15624;
  TCCR1B |= (1 << WGM12);                 // mod CTC
  TCCR1B |= (1 << CS12) | (1 << CS10);    // prescaler 1024
  TIMSK1 |= (1 << OCIE1A);                // intrerupere la compare match A
  interrupts();
}

ISR(TIMER1_COMPA_vect) {
  sec++;
  if (sec >= 60) {
    sec = 0;
    min++;
    if (min >= 60) {
      min = 0;
      ore++;
      if (ore >= 24) {
        ore = 0;
      }
    }
  }
  tickSecunda = true;
}

// ---------- Afisare ----------
void afiseaza(float temp, uint8_t h, uint8_t m, uint8_t s) {
  char linie[17];

  // Randul 1: Temp
  lcd.setCursor(0, 0);
  lcd.print("Temp = ");
  lcd.print((int)(temp + 0.5));           // rotunjit la grad
  lcd.print("'C   ");

  // Randul 2: Ora 
  snprintf(linie, sizeof(linie), "Ora %02u:%02u:%02u", h, m, s);
  lcd.setCursor(0, 1);
  lcd.print(linie);
}

void setup() {
  lcd.begin(16, 2);
  adcInit();
  timer1Init();

  // Afisare imediata la pornire
  afiseaza(citesteTemperatura(), ore, min, sec);
}

void loop() {
  if (tickSecunda) {
    noInterrupts();
    tickSecunda = false;
    uint8_t h = ore, m = min, s = sec;
    interrupts();

    afiseaza(citesteTemperatura(), h, m, s);
  }
}
